\section{Group Action Definition}

\begin{frame}
	\framesubtitle{GroupAction Class Definition}

	\begin{definition}[Group Action]
		A group action of a group $G$ on a set $X$ is a map $\cdot : G \times X \to X$ satisfying:
		\begin{itemize}
			\item (Compatibility) $(g_1 g_2)\cdot x = g_1\cdot(g_2\cdot x)$
			\item (Identity) $1\cdot x = x$
		\end{itemize}
	\end{definition}

	\vspace{0.5cm}

\section{Faithful and Transitive Actions}

\begin{frame}
	\framesubtitle{Faithful and Transitive Actions}

	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\textbf{Faithful Action}\\
			An action is faithful if distinct group elements induce distinct permutations:
			\[
				(\forall x \in X,\; g_1 \cdot x = g_2 \cdot x) \implies g_1 = g_2
			\]

			\leancodefile[
				firstline=25,
				lastline=26,
				firstnumber=25
			]{../../lean/GroupAction/Defs.lean}{https://github.com/FrankieeW/GroupAction/blob/main/lean/GroupAction/Defs.lean}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{Transitive Action}\\
			An action is transitive if any two elements can be connected:
			\[
				\forall x_1, x_2 \in X,\; \exists g \in G \text{ s.t. } g \cdot x_1 = x_2
			\]

			\leancodefile[
				firstline=33,
				lastline=40,
				firstnumber=33
			]{../../lean/GroupAction/Defs.lean}{https://github.com/FrankieeW/GroupAction/blob/main/lean/GroupAction/Defs.lean}
		\end{column}
	\end{columns}
\end{frame}

\section{Permutation Representation}

\begin{frame}
	\framesubtitle{Permutation Representation}

	For each $g \in G$, define $\sigma_g : X \to X$ by $\sigma_g(x) = g \cdot x$. This yields the permutation representation $\phi : G \to \mathrm{Sym}(X)$.

\section{Stabilizer Subgroup}

\begin{frame}
	\framesubtitle{Stabilizer Subgroup}

	For $x \in X$, the stabilizer is $G_x = \{\, g \in G \mid g \cdot x = x \,\}$.

	% \leancodefile[
	% 	firstline=24,
	% 	lastline=25,
	% 	firstnumber=24
	% ]{../../lean/GroupAction/Stabilizer.lean}{https://github.com/FrankieeW/GroupAction/blob/main/lean/GroupAction/Stabilizer.lean}

	\textbf{Theorem:} $G_x$ is a subgroup of $G$.
\end{frame}

\section{Examples}

\begin{frame}
	\framesubtitle{Symmetric Group Action}

	The symmetric group $\mathrm{Sym}(X)$ acts on $X$ by evaluation: $\sigma \cdot x := \sigma(x)$.
